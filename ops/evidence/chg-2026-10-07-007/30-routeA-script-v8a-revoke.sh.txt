#!/usr/bin/env bash
# Bot revokes its own token id 41. Requires DELETE 204, then proves the token is dead, then truncates local copies.
# Prints HTTP codes and token ids/names only.
set -euo pipefail
umask 077
S=$(cd "$(dirname "$0")/.." && pwd)
E="$S/v8a"; mkdir -p "$E"; TS=$(date -u +%Y%m%dT%H%M%SZ)
DEL=$("$S/bin/forge-as-bot.sh" DELETE /api/v1/users/wires-publisher/tokens/41 | tail -1)
echo "DELETE tokens/41 -> $DEL" | tee "$E/revoke-$TS.txt"
[ "$DEL" = "HTTP 204" ] || { echo "STOP DELETE not 204, nothing emptied"; exit 2; }
U=$("$S/bin/forge-as-token.sh" /api/v1/user)
echo "token $U" | tee -a "$E/revoke-$TS.txt"
V=$(curl -sS -o /dev/null -w '%{http_code}' -K <(printf 'user = "wires-publisher:%s"\n' "$(head -1 "$S/auth/bot.token")") \
  'https://forge.bluejungle.net/v2/token?service=container_registry&scope=repository:wires-publisher/cluster-wires:pull')
echo "token /v2/token -> $V" | tee -a "$E/revoke-$TS.txt"
L=$("$S/bin/forge-as-bot.sh" GET /api/v1/users/wires-publisher/tokens | sed '$d' | jq -c '[.[] | {id, name}]')
echo "token list -> $L" | tee -a "$E/revoke-$TS.txt"
case "$U" in *'-> 401') ;; *) echo "STOP /api/v1/user not 401"; exit 3;; esac
[ "$V" = 401 ] || { echo "STOP /v2/token not 401"; exit 4; }
echo "$L" | jq -e 'all(.[]; .id != 41)' >/dev/null || { echo "STOP id 41 still listed"; exit 5; }
: > "$S/auth/bot.token"; : > "$S/auth/wires-publisher-auth.json"
stat -f '%N %z' "$S/auth/bot.token" "$S/auth/wires-publisher-auth.json" | sed "s|$S/||" | tee -a "$E/revoke-$TS.txt"
echo "OK token 41 dead, local copies emptied"
